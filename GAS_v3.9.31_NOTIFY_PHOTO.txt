2) 新函式：
function tgSendPhoto_(chatId, b64, caption) {
  b64 = String(b64 || '').replace(/^data:image\/\w+;base64,/, '');
  if (!b64 || b64.length > 9000000) return { ok: false, error: 'photo missing or too large' };
  const mime = b64.slice(0, 4) === '/9j/' ? 'image/jpeg' : 'image/png';
  const blob = Utilities.newBlob(Utilities.base64Decode(b64), mime, 'summary.' + (mime === 'image/jpeg' ? 'jpg' : 'png'));
  const res = UrlFetchApp.fetch(TG() + '/sendPhoto', { method: 'post', muteHttpExceptions: true, payload: { chat_id: String(chatId), caption: String(caption || '').slice(0, 1000), photo: blob } });
  return JSON.parse(res.getContentText());
}
